  ❌ BEFORE (RACE CONDITION):

         authRepository.currentUser.collect { firebaseUser ->
             if (firebaseUser != null) {
                 loadCurrentUser()              // ❌ One-time fetch
                 startRealtimeUserListener()    // ❌ Real-time listener
                 // Race condition: Which updates _currentUser first?
             }
         }
     }

  ✅ AFTER (SINGLE SOURCE OF TRUTH):

                 startRealtimeUserListener()    // ✅ Only real-time listener
                 // No race condition - listener is the single source of truth
             }
         }
     }
