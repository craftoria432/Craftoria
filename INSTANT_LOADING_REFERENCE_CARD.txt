import com.gcuf.craftoria.utils.ScreenLoadingOptimizer
import com.gcuf.craftoria.utils.ScreenTransitionOptimizer

┌───────────────────────────────────────────────────────────────────────────┐
│ 🔧 VIEWMODEL SETUP (3 lines)                                             │
└───────────────────────────────────────────────────────────────────────────┘
private lateinit var optimizer: ScreenLoadingOptimizer<List<YourType>>

val data: StateFlow<List<YourType>> get() = optimizer.data
val isFirstLoad: StateFlow<Boolean> get() = optimizer.isFirstLoad
val isRefreshing: StateFlow<Boolean> get() = optimizer.isRefreshing

┌───────────────────────────────────────────────────────────────────────────┐
│ 📥 LOAD DATA (5 lines)                                                   │
└───────────────────────────────────────────────────────────────────────────┘
fun loadData(userId: String) {
    viewModelScope.launch {
        optimizer = ScreenLoadingOptimizer(
            cacheKey = "my_data_$userId",
            emptyValue = emptyList()
        )
        optimizer.loadData {
            try {
                Result.success(repository.getData(userId))
            } catch (e: Exception) {
                Result.failure(e)
            }
        }
    }
}

┌───────────────────────────────────────────────────────────────────────────┐
│ 🎨 SCREEN UI (Pattern 1: Simple)                                         │
└───────────────────────────────────────────────────────────────────────────┘
val data by viewModel.data.collectAsState()
val isFirstLoad by viewModel.isFirstLoad.collectAsState()
val isRefreshing by viewModel.isRefreshing.collectAsState()

LaunchedEffect(userId) { viewModel.loadData(userId) }

when {
    isFirstLoad && data.isEmpty() -> LoadingScreen()
    data.isEmpty() -> EmptyStateScreen()
    else -> ContentList(data)
}

┌───────────────────────────────────────────────────────────────────────────┐

Scaffold(
    topBar = {
        Column {
            TopAppBar(...)

                LinearProgressIndicator(Modifier.fillMaxWidth())
            }
        }
    }
) { padding ->
    Box(Modifier.fillMaxSize().padding(padding)) {
        when {
            isFirstLoad && data.isEmpty() -> LoadingScreen()
            data.isEmpty() -> EmptyStateScreen()
            else -> ContentList(data)
        }
    }
}

┌───────────────────────────────────────────────────────────────────────────┐
│ 🔄 FORCE REFRESH (Pull-to-Refresh)                                       │
└───────────────────────────────────────────────────────────────────────────┘
fun refreshData() {
    viewModelScope.launch {
        optimizer.loadData(forceRefresh = true) {
            Result.success(repository.getData())
        }
    }
}

┌───────────────────────────────────────────────────────────────────────────┐
│ 📊 LOADING STATE DECISION TREE                                           │
└───────────────────────────────────────────────────────────────────────────┘

Is this the first visit?
    ├─ YES (isFirstLoad = true)
    │   └─ Is data empty?
    │       ├─ YES → Show full loading indicator ⏳
    │       └─ NO  → Show content (shouldn't happen)
    │
    └─ NO (isFirstLoad = false)
        └─ Is refreshing?
            ├─ YES (isRefreshing = true)
            │   └─ Show content + subtle progress bar ▓▓▓▓▓▓
            │
            └─ NO (isRefreshing = false)
                └─ Is data empty?
                    ├─ YES → Show empty state 📭
                    └─ NO  → Show content 📦

┌───────────────────────────────────────────────────────────────────────────┐
│ 🐛 COMMON ISSUES & FIXES                                                 │
└───────────────────────────────────────────────────────────────────────────┘
Issue: Data not showing instantly
Fix:   Check cache key is consistent (use "data_$userId" not just "data")

Issue: Loading shows every time
Fix:   Use isFirstLoad, not a custom loading state

Issue: Stale data showing
Fix:   Reduce cacheDurationMs or call clearCache()

Issue: Memory issues
Fix:   Cache auto-expires, or call clearAllCaches() on logout
