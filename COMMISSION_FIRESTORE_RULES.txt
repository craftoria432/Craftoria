// ============================================================================
// ADMIN COMMISSIONS COLLECTION
// ============================================================================
// Stores individual commission records for each order
// Only admins can read, only super_admin can write

match /admin_commissions/{document=**} {
  // ✅ Admins can read all commissions
  allow read: if request.auth != null && 
              request.auth.token.role in ['super_admin', 'admin'];
  
  // ✅ Only super_admin can create/update/delete
  allow create: if request.auth != null && 
                request.auth.token.role == 'super_admin' &&
                request.resource.data.commission_amount > 0 &&
                request.resource.data.status in ['pending', 'processing', 'paid', 'failed'];
  
  allow update: if request.auth != null && 
                request.auth.token.role == 'super_admin' &&
                request.resource.data.commission_amount == resource.data.commission_amount;
  
  allow delete: if request.auth != null && 
                request.auth.token.role == 'super_admin';
}

// ============================================================================
// ADMIN EARNINGS COLLECTION
// ============================================================================
// Aggregated earnings summary
// Only admins can read, only system can write

match /admin_earnings/{document=**} {
  // ✅ Admins can read earnings summary
  allow read: if request.auth != null && 
              request.auth.token.role in ['super_admin', 'admin'];
  
  // ✅ Only system (cloud functions) can write
  // This is enforced by having only backend code update this
  allow write: if false;  // Disabled for security
}

// ============================================================================
// COMMISSION SETTINGS COLLECTION
// ============================================================================
// Configuration for commission system
// All authenticated users can read, only super_admin can write

match /commission_settings/{document=**} {
  // ✅ All authenticated users can read settings
  allow read: if request.auth != null;
  
  // ✅ Only super_admin can update settings
  allow create: if request.auth != null && 
                request.auth.token.role == 'super_admin' &&
                request.resource.data.commission_rate >= 0 &&
                request.resource.data.commission_rate <= 100 &&
                request.resource.data.payment_settlement_days > 0;
  
  allow update: if request.auth != null && 
                request.auth.token.role == 'super_admin' &&
                request.resource.data.commission_rate >= 0 &&
                request.resource.data.commission_rate <= 100 &&
                request.resource.data.payment_settlement_days > 0;
  
  allow delete: if false;  // Never delete settings
}

// ============================================================================
// SELLER PAYMENTS COLLECTION - COMMISSION FIELDS
// ============================================================================
// Updated to include commission tracking
// Existing rules apply, with new commission fields

match /seller_payments/{document=**} {
  // ✅ Sellers can read their own payments
  allow read: if request.auth != null && 
              (request.auth.uid == resource.data.seller_id ||
               request.auth.uid in resource.data.involved_seller_ids ||
               request.auth.token.role in ['super_admin', 'admin']);
  
  // ✅ Only system (cloud functions) can create/update
  allow write: if false;  // Disabled for security
}

// ============================================================================
// HELPER FUNCTIONS
// ============================================================================

function isAdmin() {
  return request.auth != null && 
         request.auth.token.role in ['super_admin', 'admin'];
}

function isSuperAdmin() {
  return request.auth != null && 
         request.auth.token.role == 'super_admin';
}

function isValidCommissionRate(rate) {
  return rate >= 0 && rate <= 100;
}

function isValidCommissionStatus(status) {
  return status in ['pending', 'processing', 'paid', 'failed'];
}

// ============================================================================
// IMPLEMENTATION NOTES
// ============================================================================

/*
1. ADMIN COMMISSIONS
   - Only admins can view commission records
   - Only super_admin can create/modify
   - Commission amount must be positive
   - Status must be valid

2. ADMIN EARNINGS
   - Read-only for admins
   - Updated only by backend (cloud functions)
   - Never deleted

3. COMMISSION SETTINGS
   - Readable by all authenticated users
   - Writable only by super_admin
   - Commission rate must be 0-100%
   - Settlement days must be positive
   - Settings are never deleted

4. SELLER PAYMENTS
   - Sellers can only read their own payments
   - Commission fields are included
   - Only backend can create/update

5. SECURITY PRINCIPLES
   - Principle of least privilege
   - Admin operations require super_admin role
   - Data validation at Firestore level
   - Immutable commission amounts
   - Audit trail via timestamps
*/
