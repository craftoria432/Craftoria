  VALIDATION LOGIC (BuyerRefundRequestScreen.kt)
═══════════════════════════════════════════════════════════════════════════════

LaunchedEffect(orderId) {
    // Get ALL refunds for this order
    val allRefunds = refundRepository.getRefundsByOrderId(orderId)
    
    // Get MOST RECENT refund (by requested_at timestamp)
    val mostRecentRefund = allRefunds.maxByOrNull { it.getRequestedAtLong() }
    
    if (mostRecentRefund != null) {
        when {
            // ❌ BLOCK: Final decision reached
            mostRecentRefund.finalDecision -> {
                errorMessage = "FINAL DECISION - No more requests"
                existingRefund = mostRecentRefund  // Show status card
            }
            
            // ❌ BLOCK: Request already pending
            mostRecentRefund.status in ["requested", "processing", ...] -> {
                errorMessage = "Request already pending"
                existingRefund = mostRecentRefund  // Show status card
            }
            
            // ❌ BLOCK: Already refunded
            mostRecentRefund.status == "completed" -> {
                errorMessage = "Already refunded"
                existingRefund = mostRecentRefund  // Show status card
            }
            
            // ✅ ALLOW: First rejection, can resubmit
            mostRecentRefund.status in ["rejected_by_seller", "rejected_by_admin"] 
            && mostRecentRefund.canResubmit -> {
                existingRefund = null  // Clear to show form
                // Show resubmission message in UI
            }
            
            // ❌ BLOCK: Rejected but cannot resubmit (shouldn't happen if finalDecision is set correctly)
            else -> {
                errorMessage = "Cannot submit refund request"
                existingRefund = mostRecentRefund
            }
        }
    } else {
        // ✅ No existing refunds - show form
        existingRefund = null
    }
}

  KEY TAKEAWAYS

✅ WHAT WAS FIXED:
  • Checks ALL refunds for order (not just first one)
  • Gets MOST RECENT refund to check current status
  • Enforces final_decision flag at UI level
  • Prevents creating new refund documents to bypass limit
  • Shows clear error messages explaining why blocked

❌ WHAT WAS BROKEN BEFORE:
  • Only checked if A refund existed (not which one)
  • Didn't check final_decision flag
  • Buyer could create infinite new refund documents
  • No order-level enforcement

🎯 RESULT:
  • Maximum 2 attempts per order (initial + 1 resubmission)
  • Permanent block after 2 rejections
  • Clear user messaging
  • Abuse-proof system
