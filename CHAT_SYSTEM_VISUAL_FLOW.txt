│     └─ chatRepository.getOrCreateChat()                                     │

│     └─ syncParticipantRoles(chatId, [user_a, user_b])                      │

│     ├─ user_a → "SELLER" (updated!)                                        │

│                                                                             │
│  5. Update chat document:                                                   │
│     ├─ participant_roles: {user_a: "SELLER", user_b: "SELLER"}            │

│                                                                             │
│  6. ChatViewModel.loadChat() loads new values:                             │
│     ├─ _chatType.value = "seller_seller"                                   │
│     └─ _otherUserRole.value = UserRole.SELLER                              │
│                                                                             │
│  7. ChatScreen recalculates canViewProfile:                                │
│     ├─ chatType = "seller_seller"                                          │
│     └─ canViewProfile = true (seller can view seller)                      │
│                                                                             │
│  RESULT:                                                                    │
│  ├─ Chat type updated: buyer_seller → seller_seller                        │
│  ├─ Profile viewing enabled: true                                          │
│  └─ UI adapts automatically                                                │
│                                                                             │
└─────────────────────────────────────────────────────────────────────────────┘

╚════════════════════════════════════════════════════════════════════════════════╝

canViewProfile = when (chatType) {
    "seller_seller" → true      // Sellers can view other sellers
    "buyer_seller" → !isCurrentUserSeller  // Only buyers view sellers
    "buyer_buyer" → true        // Buyers can view other buyers
    else → false
}

val receiptText = when {
    message.isRead → "✓✓"           // Double tick = read
    message.deliveredAt > 0 → "✓"   // Single tick = delivered
    else → ""                        // No tick = sent
}

val receiptColor = when {
    message.isRead → Primary         // Blue for read
    message.deliveredAt > 0 → TextSecondary  // Gray for delivered
    else → TextLight                 // Light gray for sent
}

TIMELINE:

Time 0ms:  Message sent
           ├─ isRead: false
           ├─ deliveredAt: 0
           └─ Display: (nothing)

Time 100ms: Chat opened by recipient
           ├─ isRead: false
           ├─ deliveredAt: 1234567895
           └─ Display: ✓ (gray)

Time 200ms: Message automatically marked as read
           ├─ isRead: true
           ├─ deliveredAt: 1234567895
           └─ Display: ✓✓ (blue)
